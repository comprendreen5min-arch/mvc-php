DataTables Server-Side Processing

Guides on setting up AJAX-based table pagination with column joins, sorting, and search filtering.

1. Model Datatable Query (with Table Joins)

Implement the join count and query statements inside your Model subclass using the helper method getDataTableDataWithJoin():

public function getProductsDataTable($start, $length, $searchValue, $orderColumn, $orderDir)
{
    $query = "
        SELECT 
            pp.id AS partners_product_id,
            pp.product_name,
            p.product_category,
            p.product_photo
        FROM partners_products pp
        LEFT JOIN products p ON pp.product_id = p.id
        WHERE pp.soft_delete = 1
    ";
    
    $countQuery = "
        SELECT COUNT(*) as total
        FROM partners_products pp
        LEFT JOIN products p ON pp.product_id = p.id
        WHERE pp.soft_delete = 1
    ";
    
    $searchColumns = [
        'pp.product_name',
        'p.product_category'
    ];
    
    return $this->getDataTableDataWithJoin(
        $query,
        $countQuery,
        $searchColumns,
        $orderColumn,
        $orderDir,
        $start,
        $length,
        $searchValue
    );
}

2. Controller Handler (Returning JSON)

Handle the request parameters, parse order columns, fetch the rows, and output in the standard JSON structure expected by DataTables:

public function all_products()
{
    if (REQUEST !== 'POST') {
        echo json_encode(['error' => 'Invalid request']);
        exit;
    }
    
    $draw = isset($_POST['draw']) ? intval($_POST['draw']) : 1;
    $start = isset($_POST['start']) ? intval($_POST['start']) : 0;
    $length = isset($_POST['length']) ? intval($_POST['length']) : 10;
    $searchValue = isset($_POST['search']['value']) ? trim($_POST['search']['value']) : '';
    
    $orderColumnIndex = isset($_POST['order'][0]['column']) ? intval($_POST['order'][0]['column']) : 0;
    $orderDir = isset($_POST['order'][0]['dir']) ? $_POST['order'][0]['dir'] : 'asc';
    
    $orderableColumns = [
        0 => 'pp.product_name',
        1 => 'p.product_category',
        2 => 'pp.id'
    ];
    
    $orderColumn = $orderableColumns[$orderColumnIndex] ?? 'pp.id';
    $orderDir = strtoupper($orderDir) === 'DESC' ? 'DESC' : 'ASC';
    
    $result = $this->mod->getProductsDataTable($start, $length, $searchValue, $orderColumn, $orderDir);
    
    echo json_encode([
        'draw' => $draw,
        'recordsTotal' => $result['totalRecords'],
        'recordsFiltered' => $result['filteredRecords'],
        'data' => $result['data']
    ]);
    exit;
}

3. Module-Based Frontend jQuery Configuration

To avoid polluting the global scope, organize your DataTable scripts using IIFEs and register them inside a global module namespace (e.g. window.DataTableModules):

Example: public/javascript/modules/datatables/_users_table.js

Handles server-side rendering, loading animation spinners, color badges mapping, and secure CRUD buttons (Edit, Reset Password, Toggle Status, Delete):

(function($) {
    'use strict';
    
    function initUsersDataTable() {
        let table = $('#all-users').DataTable({
            "processing": true,
            "serverSide": true,
            "pageLength": 10,
            "lengthMenu": [[10, 25, 50, 100], [10, 25, 50, 100]],
            "order": [[5, 'asc']],
            "language": {
                "processing": '<i class="fa fa-spinner fa-spin fa-3x fa-fw"></i><span class="sr-only">Loading...</span>',
                "emptyTable": "No users found",
                "zeroRecords": "No matching users found"
            },
            "ajax": {
                "url": jsConfig.URLROOT + "/users/all_users",
                "type": "POST",
                "error": function(xhr, error, thrown) {
                    console.error('DataTables error:', error, thrown);
                    alert('Error loading data. Please refresh the page.');
                }
            },
            "columns": [
                { "data": "first_name" },
                { "data": "last_name" },
                { "data": "email" },
                { "data": "phone" },
                { 
                    "data": "user_role",
                    "render": function(data, type, row) {
                        let roles = {
                            1: '<span class="badge bg-primary px-2 py-1">Admin</span>',
                            2: '<span class="badge bg-lavender px-2 py-1">Manager</span>',
                            3: '<span class="badge bg-peach px-2 py-1">Partner</span>',
                            4: '<span class="badge bg-lightgreen px-2 py-1">Transporter</span>',
                            5: '<span class="badge bg-mint px-2 py-1">Driver</span>',
                            6: '<span class="badge bg-sky px-2 py-1">Pilot</span>',
                            7: '<span class="badge bg-softgray px-2 py-1">Accounter</span>',
                            8: '<span class="badge bg-darktext px-2 py-1">Greeter</span>'
                        };
                        return roles[data] || '<span class="badge bg-secondary">Unknown</span>';
                    },
                    "className": "text-center"
                },
                { 
                    "data": "user_status",
                    "render": function(data, type, row) {
                        return data == 1 
                            ? '<label class="badge bg-success py-1 px-2">Active</label>' 
                            : '<label class="badge bg-danger py-1 px-2">Inactive</label>';
                    },
                    "className": "text-center"
                },
                {
                    "data": "id",
                    "orderable": false,
                    "searchable": false,
                    "render": function(data, type, row) {
                        if(data == 1){
                            return '<span class="badge bg-pink px-2 py-1"><i class="fa fa-ban"></i>  Super Admin</span>';
                        }
                        return `
                            <div class="me-2" role="group">
                                <a href="${jsConfig.URLROOT}/users/edit/${data}" class="btn btn-sm btn-primary" title="Edit">
                                    <i class="fa fa-edit"></i>
                                </a>
                                <a href="${jsConfig.URLROOT}/users/password/${data}" class="btn btn-sm btn-warning" title="Change Password">
                                    <i class="fa fa-key"></i>
                                </a>
                                <form method="POST" action="${jsConfig.URLROOT}/users/user_status/${data}" style="display:inline;" onsubmit="return confirm('Toggle status?');">
                                    <button type="submit" class="btn btn-sm btn-info" title="Toggle Status">
                                        <i class="fa fa-toggle-on"></i>
                                    </button>
                                </form>
                                <form method="POST" action="${jsConfig.URLROOT}/users/delete/${data}" style="display:inline;" onsubmit="return confirm('Are you sure?');">
                                    <button type="submit" class="btn btn-sm btn-danger" title="Delete">
                                        <i class="fa fa-trash"></i>
                                    </button>
                                </form>
                            </div>
                        `;
                    },
                    "className": "text-center"
                }
            ]
        });
    }
    
    // Register into modules registry
    window.DataTableModules = window.DataTableModules || {};
    window.DataTableModules.users = {
        init: function() {
            initUsersDataTable();
        }
    };
})(jQuery);
